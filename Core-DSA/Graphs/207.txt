class Solution {
    public boolean canFinish(int numCourses, int[][] prerequisites) {
        
        if(prerequisites.length==0) return true;

        int visited[] = new int[numCourses];
        Map<Integer, List<Integer>> adjacency = new HashMap<>();

        for(int i=0; i<prerequisites.length;i++)
        {
            adjacency.computeIfAbsent(prerequisites[i][1], ls -> new ArrayList<>()).add(prerequisites[i][0]);
        }

        for(int i=0; i<numCourses; i++)
        {
            if(!dfsHelper(i,adjacency, visited)) return false;
        }

        return true;

    }

    private boolean dfsHelper(int node, Map<Integer, List<Integer>> adjacency, int[] visited)
    {
        if(visited[node]==1) return false;
        if(visited[node]==2) return true;
        
        visited[node]=1; // mark the node as visiting...
        for(int n : adjacency.getOrDefault(node, new ArrayList<>()))
        {
            if(!dfsHelper(n,adjacency,visited)) return false;
        }

        visited[node]=2; // mark the node as completed...
        return true;
    }
}

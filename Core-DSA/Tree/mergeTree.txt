/**
 * Definition for a binary tree node.
 * public class TreeNode {
 *     int val;
 *     TreeNode left;
 *     TreeNode right;
 *     TreeNode() {}
 *     TreeNode(int val) { this.val = val; }
 *     TreeNode(int val, TreeNode left, TreeNode right) {
 *         this.val = val;
 *         this.left = left;
 *         this.right = right;
 *     }
 * }
 *  https://neetcode.io/problems/merge-two-binary-trees/question
 */


class Solution {
    public TreeNode mergeTrees(TreeNode root1, TreeNode root2) {

        return helperMerge(root1, root2);
        
    }

    private TreeNode helperMerge(TreeNode root1, TreeNode root2)
    {
        if(root1==null && root2==null) return null;
        if(root1==null) return root2;
        if(root2==null) return root1;
        root1.val+=root2.val;
        root1.left =helperMerge(root1.left, root2.left);
        root1.right =helperMerge(root1.right, root2.right);
        return root1;
    }
}
